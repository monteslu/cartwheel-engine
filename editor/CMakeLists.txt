# The cartwheel editor: a cartwheel app built with the engine's UI.
#   core/       the editor core (commands, history, documents, dock, shell)
#   modules/X/  one directory per feature; every .c inside is compiled and the
#               module registry is GENERATED from the directory list, so a new
#               module edits no shared file (see include/cwed/editor.h)
#   app/        the app entry (editor, editor_session)
#   tests/      command-level tests (unit_editor) and the scripted session
file(GLOB CWED_CORE_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/core/*.c)
file(GLOB CWED_MODULE_DIRS LIST_DIRECTORIES true CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/modules/*)
list(SORT CWED_MODULE_DIRS)
set(CWED_MODULE_SRCS "")
set(_externs "")
set(_table "")
set(_names "")
set(_count 0)
foreach(dir ${CWED_MODULE_DIRS})
  if(IS_DIRECTORY ${dir})
    get_filename_component(mod ${dir} NAME)
    if(NOT mod MATCHES "^[a-z][a-z0-9_]*$")
      message(FATAL_ERROR "editor module directory '${mod}': names are lower_case C identifiers")
    endif()
    file(GLOB_RECURSE _srcs CONFIGURE_DEPENDS ${dir}/*.c)
    list(APPEND CWED_MODULE_SRCS ${_srcs})
    string(APPEND _externs "extern const cwed_module cwed_module_${mod};\n")
    string(APPEND _table "  &cwed_module_${mod},\n")
    string(APPEND _names "  \"${mod}\",\n")
    math(EXPR _count "${_count} + 1")
  endif()
endforeach()
set(_gen ${CMAKE_CURRENT_BINARY_DIR}/editor_modules.c)
set(_content "/* GENERATED by editor/CMakeLists.txt from the editor/modules directories: do not edit. */\n#include \"cwed/editor.h\"\n${_externs}const cwed_module *const cwed__modules[] = {\n${_table}  0\n};\nconst char *const cwed__module_names[] = {\n${_names}  0\n};\nconst int cwed__module_count = ${_count};\n")
if(EXISTS ${_gen})
  file(READ ${_gen} _old)
else()
  set(_old "")
endif()
if(NOT _old STREQUAL _content)
  file(WRITE ${_gen} "${_content}")
endif()

add_library(cartwheel_editor STATIC ${CWED_CORE_SRCS} ${CWED_MODULE_SRCS} ${_gen})
target_include_directories(cartwheel_editor PUBLIC include PRIVATE core)
target_compile_options(cartwheel_editor PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
target_link_libraries(cartwheel_editor PUBLIC cartwheel cw_importers) # cw_importers: the import module runs cwpack's importers
# the editor's three.c program table (cmake/programs.cmake: its views, gizmos,
# previews and demo project), registered by cwed_init
cw_program_table(${CMAKE_CURRENT_SOURCE_DIR}/programs/cw_editor.txt _t)
target_link_libraries(cartwheel_editor PUBLIC cwprog_cw_editor)

set(CWED_DEMO ${CMAKE_CURRENT_SOURCE_DIR}/demo_project)
# The editor's bundled files: the demo project plus the project
# templates under cwtemplates/ (what project.new copies in a cart), staged as
# symlinks so edits show without a reconfigure, and cwtemplates.txt listing
# every template file (a cart cannot list its assets).
set(CWED_ASSETS ${CMAKE_CURRENT_BINARY_DIR}/editor_assets)
file(MAKE_DIRECTORY ${CWED_ASSETS})
file(GLOB _cwed_demo RELATIVE ${CWED_DEMO} ${CWED_DEMO}/*)
foreach(f ${_cwed_demo})
  file(CREATE_LINK ${CWED_DEMO}/${f} ${CWED_ASSETS}/${f} SYMBOLIC)
endforeach()
file(CREATE_LINK ${CMAKE_CURRENT_SOURCE_DIR}/templates ${CWED_ASSETS}/cwtemplates SYMBOLIC)
file(GLOB_RECURSE _cwed_tpl RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}/templates CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/templates/*)
list(SORT _cwed_tpl)
list(FILTER _cwed_tpl EXCLUDE REGEX "(^|/)\\.") # wasmcart-pack skips dot files
string(REPLACE ";" "\n" _cwed_tpl "${_cwed_tpl}")
file(CONFIGURE OUTPUT ${CWED_ASSETS}/cwtemplates.txt CONTENT "${_cwed_tpl}\n" @ONLY)
# natively the editor runs the engine checkout's tools (tools/cwproject.mjs,
# tools/cwexport.mjs) and copies templates from editor/templates;
# CARTWHEEL_DIR overrides where that checkout is
if(NOT CW_CART)
  target_compile_definitions(cartwheel_editor PRIVATE CWED_ENGINE_DIR="${CMAKE_SOURCE_DIR}")
endif()
set(CWED_DEMO ${CWED_ASSETS})
set_property(GLOBAL PROPERTY CWED_ASSETS_DIR ${CWED_ASSETS}) # project editors (cmake/plugins.cmake) bundle its cwtemplates
# a project build needs only the library: its <app>_editor is the app
if(CWED_LIBRARY_ONLY)
  return()
endif()
cw_add_app(editor SOURCES app/main.c ASSETS ${CWED_DEMO} GL_ONLY)
target_link_libraries(editor PRIVATE cartwheel_editor)
cw_add_app(editor_session SOURCES app/main.c ASSETS ${CWED_DEMO} GL_ONLY)
target_link_libraries(editor_session PRIVATE cartwheel_editor)
target_compile_definitions(editor_session PRIVATE EDITOR_SESSION=1)
cw_add_app(unit_editor SOURCES tests/unit_editor.c ASSETS ${CWED_DEMO} GL_ONLY)
target_link_libraries(unit_editor PRIVATE cartwheel_editor)
target_include_directories(unit_editor PRIVATE core ${CMAKE_SOURCE_DIR}/tests/unit)

# editor_import (editor/tests/import_app.c): the editor with a
# scripted "drag a glTF in, see it imported, place it" session. The CC0 test
# files in editor/tests/import (gen-assets.mjs) are compiled in, so the cart
# brings them into its project the way an archive import does.
file(GLOB CWED_IMPORT_FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/import/*)
list(SORT CWED_IMPORT_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CWED_IMPORT_FILES})
set(_imp_c "/* GENERATED from editor/tests/import by editor/CMakeLists.txt: do not edit. */\ntypedef struct { const char *name; const unsigned char *data; unsigned len; } embedded;\n")
set(_imp_tab "")
set(_i 0)
foreach(f ${CWED_IMPORT_FILES})
  get_filename_component(_n ${f} NAME)
  file(READ ${f} _hex HEX)
  file(SIZE ${f} _sz)
  string(REGEX REPLACE "(..)" "0x\\1," _hex "${_hex}")
  string(APPEND _imp_c "static const unsigned char f${_i}[] = {${_hex}0};\n")
  string(APPEND _imp_tab "  { \"${_n}\", f${_i}, ${_sz} },\n")
  math(EXPR _i "${_i} + 1")
endforeach()
string(APPEND _imp_c "const embedded cwed_import_files[] = {\n${_imp_tab}  { 0, 0, 0 }\n};\nconst int cwed_import_file_count = ${_i};\n")
set(_imp_gen ${CMAKE_CURRENT_BINARY_DIR}/import_files.c)
if(EXISTS ${_imp_gen})
  file(READ ${_imp_gen} _old)
else()
  set(_old "")
endif()
if(NOT _old STREQUAL _imp_c)
  file(WRITE ${_imp_gen} "${_imp_c}")
endif()
cw_add_app(editor_import SOURCES tests/import_app.c ${_imp_gen} ASSETS ${CWED_DEMO} GL_ONLY)
target_link_libraries(editor_import PRIVATE cartwheel_editor)
target_include_directories(editor_import PRIVATE core)
if(NOT CW_CART)
  # the native editor dlopens code built outside it (a game's
  # node types, project plugins: runtime and editor parts), so like cwhot it
  # carries the whole engine and the whole editor and exports their symbols
  # for those libraries to link against.
  set_target_properties(editor PROPERTIES ENABLE_EXPORTS ON)
  target_link_options(editor PRIVATE "LINKER:--whole-archive" $<TARGET_FILE:cartwheel_editor> $<TARGET_FILE:cartwheel>
    $<TARGET_FILE:three> $<TARGET_FILE:box2d> $<TARGET_FILE:box3d> $<TARGET_FILE:cw_miniaudio> "LINKER:--no-whole-archive")
  target_link_libraries(editor PRIVATE dl)
endif()
